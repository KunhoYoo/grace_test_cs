권장 Rules

admin, staff, cs role 계정만 전체 `/users` 읽기/쓰기를 허용합니다.

{
  "rules": {
    "users": {
      ".read": "auth != null && (root.child('users').child(auth.uid).child('profile').child('role').val() === 'admin' || root.child('users').child(auth.uid).child('profile').child('role').val() === 'staff' || root.child('users').child(auth.uid).child('profile').child('role').val() === 'cs')",
      "$uid": {
        ".read": "auth != null && (auth.uid === $uid || root.child('users').child(auth.uid).child('profile').child('role').val() === 'admin' || root.child('users').child(auth.uid).child('profile').child('role').val() === 'staff' || root.child('users').child(auth.uid).child('profile').child('role').val() === 'cs')",
        ".write": "auth != null && (auth.uid === $uid || root.child('users').child(auth.uid).child('profile').child('role').val() === 'admin' || root.child('users').child(auth.uid).child('profile').child('role').val() === 'staff' || root.child('users').child(auth.uid).child('profile').child('role').val() === 'cs')"
      }
    }
  }
}
